# 工作流引擎优化设计方案

> 基于现有实现的问题诊断与改造方案，聚焦 LangGraph State 规范化、Agent 独立消息渲染、前端 Run Dialog 统一

## 0. 动机与问题诊断

### 当前架构三个核心缺陷

| # | 问题 | 现有代码位置 | 后果 |
|---|------|-------------|------|
| 1 | **ExecutionContext 在 State 之外管理状态** | `context.py:ExecutionContext` 被 `nodes.py` 每个 executor 通过 `self.ctx` 引用 | LangGraph 无法追踪状态变化，无法做 checkpoint；多节点并行时 node_outputs 竞争写入 |
| 2 | **所有 Agent 共享一条 workflow_message** | `nodes.py:111` — `HumanMessage(content=self.workflow_message)` 由 `engine.py:52` 统一渲染 | 每个 Agent 收到相同的 HumanMessage，无法根据上游结果定制输入；Agent 自身的 `message_template` 被浪费 |
| 3 | **WorkflowState 缺少关键字段** | `graph_builder.py:19-23` — 无 messages、无 params、node_outputs 无 reducer | 不符合 LangGraph State 设计模式，消息历史无法积累 |

### 次要问题

| # | 问题 | 位置 | 说明 |
|---|------|------|------|
| 4 | 列表页 Run Dialog 用原始 JSON 输入 | `index.vue:196-207` | 编辑器已用模板表单，列表页仍为 textarea JSON，体验不一致 |
| 5 | Agent 节点配置看不到 Agent 的 message_template | `AgentConfig.vue` | 用户无法确认最终发给 Agent 的消息内容 |

## 2. 后端改造

### 2.1 新建 `app/workflows/state_utils.py`

**来源**：从现有 `context.py:ExecutionContext`（L9-97）提取所有方法为纯函数。

**变化**：入参从 `self` 改为 `state: dict`，`input` 路径改为读 `state["params"]`。

```python
"""
工作流状态工具函数 — 纯函数，无状态
替代原 ExecutionContext，所有操作基于 WorkflowState dict
"""
import re
from typing import Any, Dict, List

VAR_PATTERN = re.compile(r"\{\{(\w[\w.]*)\}\}")


def extract_template_vars(template: str) -> List[str]:
    """提取模板中的变量名列表，如 {{stock}} → ["stock"]"""
    matches = VAR_PATTERN.findall(template)
    return list(dict.fromkeys(matches))  # 去重保序


def resolve_value(state: Dict[str, Any], path: str) -> Any:
    """
    从 state 中解析变量路径。

    支持路径:
      {{input.stock}}   → state["params"]["stock"]
      {{node_id.output}} → state["node_outputs"]["node_id"]["output"]
      {{item.name}}      → state["_loop_item"]["name"]  (循环内)

    对应原 ExecutionContext.resolve_value() (context.py L34-36)
    变化：root="input" 改为读 state["params"]
    """
    parts = path.split(".")
    if not parts:
        return None

    root = parts[0]

    # 用户输入参数: {{input.xxx}} → state["params"]
    if root == "input" and len(parts) > 1:
        return _navigate(state.get("params", {}), parts[1:])

    # 循环变量: {{item.xxx}}
    if root == "item" and "_loop_item" in state:
        return _navigate(state["_loop_item"], parts[1:])

    # 节点输出: {{node_id.output}} → state["node_outputs"]
    node_out = state.get("node_outputs", {}).get(root)
    if node_out is not None:
        if len(parts) == 1:
            return node_out
        return _navigate(node_out, parts[1:])

    return None


def resolve_mapping(state: Dict[str, Any], mapping: Dict[str, str]) -> Dict[str, Any]:
    """
    解析 input_mapping 配置。

    对应原 ExecutionContext.resolve_mapping() (context.py L18-23)
    """
    return {key: resolve_value(state, tpl) for key, tpl in mapping.items()}


def render_template(state: Dict[str, Any], template: str) -> str:
    """
    渲染模板字符串，替换 {{...}} 为 state 中的实际值。

    对应原 ExecutionContext.render_template() (context.py L25-32)
    """
    def replacer(match: re.Match) -> str:
        value = resolve_value(state, match.group(1))
        return str(value) if value is not None else match.group(0)
    return VAR_PATTERN.sub(replacer, template)


def _navigate(obj: Any, parts: List[str]) -> Any:
    """
    在嵌套 dict/list 中按路径导航。
    对应原 ExecutionContext._navigate() (context.py L81-96)，逻辑完全不变。
    """
    current = obj
    for part in parts:
        if isinstance(current, dict):
            current = current.get(part)
        elif isinstance(current, (list, tuple)):
            try:
                current = current[int(part)]
            except (ValueError, IndexError):
                return None
        else:
            return None
        if current is None:
            return None
    return current
```

### 2.2 重写 `app/workflows/graph_builder.py`

#### WorkflowState 对比

```python
# ── 旧 (graph_builder.py L19-23) ──
class WorkflowState(TypedDict):
    context: Dict[str, Any]       # 用途模糊，实际只存 {}
    node_outputs: Dict[str, Any]  # 无 reducer，多节点并行时覆盖
    current_node: str
    errors: Annotated[List[str], operator.add]

# ── 新 ──
class WorkflowState(TypedDict):
    messages: Annotated[list, add_messages]      # LangGraph 标准消息 reducer
    params: Dict[str, Any]                        # 结构化参数（用户输入）
    node_outputs: Annotated[Dict[str, Any], dict_merge]  # 带 reducer，自动合并
    errors: Annotated[List[str], operator.add]    # 不变
    iterations: int                               # 迭代计数（为辩论循环预留）
    sender: str                                   # 最近执行的节点 ID
```

字段变化说明：

| 字段 | 旧 | 新 | 变化原因 |
|------|----|----|----------|
| `messages` | 无 | `Annotated[list, add_messages]` | LangGraph 标准；各节点 AI 回复自动追加 |
| `params` | 无 | `Dict[str, Any]` | 替代 `ctx.variables["input"]`，承载用户输入 |
| `node_outputs` | `Dict`（无 reducer） | `Annotated[Dict, dict_merge]` | 多节点并行时自动合并，不互相覆盖 |
| `context` | `Dict` | 删除 | 功能由 params + node_outputs 承担 |
| `current_node` | `str` | 删除 | 语义不清，用 `sender` 替代 |
| `iterations` | 无 | `int` | 预留辩论循环支持 |
| `sender` | 无 | `str` | 追踪最近执行节点，支持消息路由 |

### 2.3 重写 `app/workflows/nodes.py`

#### IONodeExecutor → 拆为 InputIO + OutputIO

**旧 `IONodeExecutor`**（nodes.py L15-49）：
- 通过 `self.ctx` 读写数据
- Input 方向：读 `self.ctx.variables.get("input", {})` 写入 node_outputs
- Output 方向：读 `self.ctx.variables.get("output_template")` 渲染最终输出

**新 `InputIONodeExecutor`**：

```python
class InputIONodeExecutor:
    """
    工作流入口节点。

    职责：
      1. 将用户输入数据写入 state["params"]
      2. 将渲染后的消息写入 state["messages"]（首条 HumanMessage）
      3. 将输入数据写入 state["node_outputs"] 供下游引用

    替代原 IONodeExecutor(direction="input") (nodes.py L15-49)
    变化：不再依赖 ExecutionContext，直接操作 state
    """

    def __init__(self, node, workflow, on_complete=None):
        self.node = node
        self.workflow = workflow
        self.node_id = node["id"]
        self.on_complete = on_complete

    async def __call__(self, state: WorkflowState) -> dict:
        input_data = state.get("params", {})

        # 渲染工作流 message_template 作为首条消息
        # 此处仅做预览/记录用途，Agent 节点使用自身的 message_template
        message_template = self.workflow.get("message_template", "")
        if message_template:
            rendered = render_template(state, message_template)
        else:
            rendered = str(input_data)

        result = {
            "params": input_data,
            "messages": [HumanMessage(content=rendered)],
            "node_outputs": {self.node_id: {"output": input_data}},
            "sender": self.node_id,
            "iterations": state.get("iterations", 0),
            "errors": [],
        }

        if self.on_complete:
            await self.on_complete(
                state.get("_run_id", ""), self.node_id, "completed",
                output=input_data,
            )

        return result
```

```python
class OutputIONodeExecutor:
    """
    工作流出口节点。

    职责：
      1. 从 state["node_outputs"] 聚合所有节点输出
      2. 如有 output_template 则渲染，否则返回全部 node_outputs

    替代原 IONodeExecutor(direction="output") (nodes.py L30-35)
    变化：不再依赖 ExecutionContext，直接从 state 读取
    """

    def __init__(self, node, workflow, on_complete=None):
        self.node = node
        self.workflow = workflow
        self.node_id = node["id"]
        self.on_complete = on_complete

    async def __call__(self, state: WorkflowState) -> dict:
        output_template = self.workflow.get("output_template")

        if output_template:
            output = render_template(state, output_template)
        else:
            # 无模板时返回全部节点输出
            output = dict(state.get("node_outputs", {}))

        result = {
            "node_outputs": {self.node_id: {"output": output}},
            "sender": self.node_id,
            "iterations": state.get("iterations", 0),
            "errors": [],
        }

        if self.on_complete:
            await self.on_complete(
                state.get("_run_id", ""), self.node_id, "completed",
                output=output,
            )

        return result
```

```python
async def _execute_single(self, state: WorkflowState, agent_id: str, variables: Dict[str, Any]) -> str:
    """
    执行单个 Agent 调用。

    核心变化：使用 Agent 自身的 message_template 渲染独立的 HumanMessage
    而非共享的 workflow_message。

    AgentSpec.message_template 由 agent_service.build_agent() 返回
    (agent_service.py L446: message_template=doc.get("message_template", ""))
    """
    from langchain_core.messages import HumanMessage, SystemMessage
    from app.services.agent_service import agent_service

    spec = await agent_service.build_agent(agent_id, variables)

    # --- 关键改造：用 Agent 自己的 message_template ---
    if spec.message_template:
        human_content = render_template(state, spec.message_template)
    else:
        # 回退：将 variables 拼成自然语言
        human_content = self._params_to_message(variables)

    messages = [
        SystemMessage(content=spec.system_prompt),
        HumanMessage(content=human_content),
    ]

    if spec.tools:
        from langgraph.prebuilt import create_react_agent
        react_agent = create_react_agent(spec.llm, spec.tools)
        result = await react_agent.ainvoke(
            {"messages": messages},
            config={"recursion_limit": spec.parameters.max_tool_calls},
        )
        output = self._extract_output(result)
        ai_messages = result.get("messages", [])
    else:
        response = await spec.llm.ainvoke(messages)
        output = response.content
        ai_messages = [response]

    await agent_service.record_usage(agent_id)

    # AI 回复追加到 state["messages"]
    state.setdefault("_new_messages", []).extend(ai_messages)

    return output

@staticmethod
def _params_to_message(params: Dict[str, Any]) -> str:
    """当 Agent 无 message_template 时，将参数拼接为自然语言"""
    if not params:
        return "请执行任务"
    parts = [f"{k}: {v}" for k, v in params.items() if v is not None]
    return "\n".join(parts)
```

**`_execute_loop` 改造**：

```python
# 旧 (nodes.py L130-156)：用 self.ctx.resolve_mapping / self.ctx.resolve_value
# 新：用 state_utils 的纯函数，通过 state["_loop_item"] 传递循环变量

async def _execute_loop(self, state, agent_id, config):
    loop_path = config.get("loop_over", "")
    loop_source = resolve_value(state, loop_path)

    if not isinstance(loop_source, list):
        logger.warning(f"loop_over for {self.node_id} did not resolve to a list")
        return None

    parallel = config.get("parallel", False)
    max_concurrency = config.get("max_concurrency", 3)
    base_mapping = config.get("input_mapping", {})
    base_variables = resolve_mapping(state, base_mapping) if base_mapping else {}

    async def run_single(item):
        loop_state = {**state, "_loop_item": item}
        loop_vars = {**base_variables, "item": item}
        return await self._execute_single(loop_state, agent_id, loop_vars)

    if parallel:
        semaphore = asyncio.Semaphore(max_concurrency)
        async def run_with_sem(item):
            async with semaphore:
                return await run_single(item)
        return await asyncio.gather(*[run_with_sem(item) for item in loop_source])
    else:
        return [await run_single(item) for item in loop_source]
```

## 3. 数据流

### 3.1 完整执行流程

```
用户提交 input_data: {stock: "000001", date: "2024-01-15"}
    │
    ▼ engine.run()
    │ state = {params: input_data, messages: [], node_outputs: {}, ...}
    │
    ▼ Input IO Node
    │ 1. state["params"] → {stock: "000001", date: "2024-01-15"}
    │ 2. render(workflow.message_template, state) → "分析 000001 在 2024-01-15..."
    │ 3. state["messages"] += [HumanMessage("分析 000001...")]
    │ 4. state["node_outputs"]["io_in"] = {output: {stock, date}}
    │
    ▼ Agent: 行业分析
    │ 1. resolve_mapping: {stock: "{{input.stock}", date: "{{input.date}}"} → {stock: "000001", date: "2024-01-15"}
    │ 2. build_agent(agent_id, {stock, date}) → AgentSpec {
    │      system_prompt: "你是一名行业分析师...",
    │      message_template: "请分析 {{input.stock}} 所属行业的最新动态"  ← Agent 自己的模板
    │    }
    │ 3. render(spec.message_template, state) → "请分析 000001 所属行业的最新动态"
    │ 4. invoke: [SystemMsg, HumanMsg("请分析 000001 所属行业的最新动态")]
    │ 5. output: "银行业近期..."
    │ 6. state["node_outputs"]["行业分析"] = {output: "银行业近期..."}
    │ 7. state["messages"] += [AIMessage("银行业近期...")]
    │
    ▼ Agent: 财务分析
    │ 1. resolve_mapping: {stock: "{{input.stock}}"} → {stock: "000001"}
    │ 2. build_agent → AgentSpec { message_template: "分析 {{input.stock}} 的财报数据" }
    │ 3. render → "分析 000001 的财报数据"
    │ 4. invoke: [SystemMsg, HumanMsg("分析 000001 的财报数据")]
    │ 5. state["node_outputs"]["财务分析"] = {output: "..."}
    │ 7. state["messages"] += [AIMessage(...)]
    │
    ▼ Agent: 投资建议
    │ 1. resolve_mapping: {
    │      industry: "{{行业分析.output}}",
    │      finance: "{{财务分析.output}}"
    │    } → {industry: "银行业近期...", finance: "..."}
    │ 2. build_agent → AgentSpec { message_template: "结合行业分析：{{行业分析.output}}，财务数据：{{财务分析.output}}，给出投资建议" }
    │ 3. render → "结合行业分析：银行业近期...，财务数据：...，给出投资建议"
    │ 4. invoke → output
    │
    ▼ Output IO Node
    │ output_template 有值 → render_template(state, output_template)
    │ 无值 → 聚合 state["node_outputs"]
    │
    ▼ engine 提取最终输出，写入 workflow_runs
```

### 3.2 关键变量流转

```
变量路径             来源                     使用方式
─────────────────────────────────────────────────────────────────────
state["params"]      engine.run() 预填         Input IO 透传，Agent 通过 {{input.xxx}} 引用
state["messages"]    各节点追加                 LangGraph add_messages reducer 自动合并
state["node_outputs"] 各节点写入               dict_merge reducer 自动合并，通过 {{node_id.output}} 引用
state["iterations"]  当前未用                   预留给辩论循环
state["sender"]      最近执行节点               预留给消息路由
```

### 3.3 嵌套子流程

```
父工作流 Agent A → Subflow 节点
                       │
                       │ resolve_mapping → 从父 state 获取参数
                       │
                       ▼ workflow_engine.run(workflow_id, input_data=variables)
                       │   创建独立的子 State
                       │   子 Input IO → 子 Agent B → 子 Output IO
                       │
                       ▼ 子结果写入父 state["node_outputs"]["subflow_1"]
```

对应 script 改动：

```typescript
// 新增
const runForm = ref<Record<string, string>>({})
const runFreeText = ref('')

const runTemplateVars = computed<string[]>(() => {
  const tpl = selectedWf.value?.message_template || ''
  const matches = tpl.match(/\{\{(\w+)\}\}/g)
  if (!matches) return []
  return [...new Set(matches.map(m => m.replace(/\{\{|\}\}/g, '')))]
})

// 改造
function selectAndRun(wf: Workflow) {
  selectedWf.value = wf
  runForm.value = {}
  for (const v of runTemplateVars.value) runForm.value[v] = ''
  runFreeText.value = ''
  runDialogVisible.value = true
}

async function doRun() {
  if (!selectedWf.value) return
  let input: Record<string, any>
  if (runTemplateVars.value.length) {
    const missing = runTemplateVars.value.find(v => !runForm.value[v]?.trim())
    if (missing) { ElMessage.error(`参数 "${missing}" 未填写`); return }
    input = {}
    for (const v of runTemplateVars.value) input[v] = runForm.value[v]
  } else {
    if (!runFreeText.value.trim()) { ElMessage.error('请输入消息'); return }
    input = { message: runFreeText.value }
  }
  // 后续调用不变
  runLoading.value = true
  try {
    const res = await workflowsApi.run(selectedWf.value.id, input)
    if (res.success) {
      ElMessage.success('执行完成')
      runDialogVisible.value = false
      await loadRuns()
    }
  } catch (err: any) {
    ElMessage.error(err?.response?.data?.detail || '执行失败')
  } finally {
    runLoading.value = false
  }
}
```
